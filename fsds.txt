local rs = game:GetService("ReplicatedStorage")
local bufferWriter = require(rs.Modules.ByteNet.process.bufferWriter)
local packetIDs = require(rs.Modules.ByteNet.packets.packetIDs)
local oldExport = bufferWriter.export

bufferWriter.export = function(...)
    local result = oldExport(...)
    if result and result.buff and result.cursor and result.cursor > 0 then
        local buf = result.buff
        local i = 0
        while i < result.cursor do
            buffer.writeu8(buf, i, 0) -- force flag
            i = i + 1
            if i >= result.cursor then break end
            local id = buffer.readu8(buf, i)
            i = i + 1
            local packet = packetIDs[id]
            if packet and packet.reader then
                local _, size = packet.reader(buf, i)
                i = i + (size or 0)
            else
                break
            end
        end
    end
    return result
end

-- Helper function to format seconds into HH:MM:SS
local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- Robust inventory scanner that checks background data modules and UI layouts
local function getInventoryCount(itemName)
    if GameUtil and GameUtil.Data then
        local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
        if type(inv) == "table" then
            if inv[itemName] then return tonumber(inv[itemName]) or 0 end
            for _, v in pairs(inv) do
                if type(v) == "table" and (v.name == itemName or v.Name == itemName) then
                    return tonumber(v.quantity or v.amount or v.Amount or v.count or v.Count or 1) or 0
                end
            end
        end
    end
    
    local list = plr.PlayerGui:FindFirstChild("MainGui") 
        and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") 
        and plr.PlayerGui:FindFirstChild("MainGui").RightPanel:FindFirstChild("Inventory") 
        and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if list then
        local total = 0
        for _, slot in ipairs(list:GetChildren()) do
            if slot:IsA("GuiObject") and (slot.Name == itemName or slot.Name:find(itemName)) then
                local quantityImg = slot:FindFirstChild("QuantityImage")
                local textObj = (quantityImg and quantityImg:FindFirstChild("QuantityText")) or slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChildWhichIsA("TextLabel")
                if textObj then
                    total = total + (tonumber(textObj.Text:match("%d+")) or 1)
                else
                    total = total + 1
                end
            end
        end
        if total > 0 then return total end
    end
    return 0
end

local killAuraActive, killAuraConn, lastKillTime = false, nil, 0

-- Fixed Kill Aura (matches real packet format)
local function startKillAura()
    if killAuraActive then return end
    killAuraActive = true

    killAuraConn = runs.Heartbeat:Connect(function()
        if not Options.killauratoggle.Value then
            if killAuraConn then killAuraConn:Disconnect() end
            killAuraConn = nil
            killAuraActive = false
            return
        end

        if not root or not GameUtil.Data.equipped then return end

        local ct = tick()
        local cooldown = tonumber(Options.killauracooldown.Value) or 0.12
        if ct - lastKillTime < cooldown then return end

        local range = tonumber(Options.killaurarange.Value) or 20
        local targets = {}

        for _, player in ipairs(game.Players:GetPlayers()) do
            if player ~= plr and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local dist = (hrp.Position - root.Position).Magnitude
                    if dist <= range then
                        local eid = player.Character:GetAttribute("EntityID")
                        if eid then
                            table.insert(targets, {entityID = eid}) -- no buffer field
                        end
                    end
                end
            end
        end

        if #targets > 0 then
            pcall(function()
                packets.SwingTool.send({
                    entityIDs = targets,
                    cframe = root.CFrame,
                    timestamp = clock.getServerTime(true)
                })
            end)
            lastKillTime = ct
        end
    end)
end

Options.killauratoggle:OnChanged(function(state)
    if state then
        startKillAura()
    else
        killAuraActive = false
        if killAuraConn then
            killAuraConn:Disconnect()
            killAuraConn = nil
        end
    end
end)

local function placeFishTrap()
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end

    local root = char.HumanoidRootPart
    local cam = workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or root.Position

    -- Raycast down in front of the player to find water
    local look = root.CFrame.LookVector
    local origin = root.Position + look * 6 + Vector3.new(0, 8, 0)

    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = {char}
    rp.IgnoreWater = false

    local result = workspace:Raycast(origin, Vector3.new(0, -40, 0), rp)
    if not result then return end

    -- Only place if we hit water or very close to water surface
    local placePos = result.Position + Vector3.new(0, 0.5, 0)

    if packets.PlaceStructure and packets.PlaceStructure.send then
        pcall(function()
            packets.PlaceStructure.send({
                buildingName = "Fish Trap",
                cframe = CFrame.new(placePos),
                vec = placePos,
                isMobile = false,
                cursor = cursorPos
            })
        end)
    end
end

local function getFishTrapsWithFish()
    local traps = {}
    local dep = workspace:FindFirstChild("Deployables")
    if not dep or not root then return traps end

    local range = tonumber(Options.fishtraprangeslider.Value) or 250
    local charPos = root.Position

    for _, trap in ipairs(dep:GetChildren()) do
        if trap.Name == "Fish Trap" then
            local contents = trap:FindFirstChild("Contents")
            if contents then
                for _, item in ipairs(contents:GetChildren()) do
                    if item.Name == "Raw Fish" and item:GetAttribute("EntityID") then
                        local pp = trap.PrimaryPart or trap:FindFirstChildWhichIsA("BasePart")
                        if pp then
                            local dist = (pp.Position - charPos).Magnitude
                            if dist <= range then
                                table.insert(traps, {
                                    trap = trap,
                                    part = pp,
                                    entityID = item:GetAttribute("EntityID"),
                                    dist = dist
                                })
                            end
                        end
                        break
                    end
                end
            end
        end
    end

    table.sort(traps, function(a, b) return a.dist < b.dist end)
    return traps
end

                while Options.autofishtraptoggle.Value and target.trap.Parent and root do

                    end
                end
            else
                task.wait(0.8)
            end
        else
            task.wait(0.4)
        end
        task.wait(0.1)
    end
end)

-- Resource Aura (optimized)
local resourceActive, resourceConn = false, nil
local lastResourceTime, lastResourceScan = 0, 0
local cachedTargets = {}

local resourceKeywords = {"Node", "Stone", "Lode", "Rock", "Coal", "Guardian", "Chunk", "Ore"}

local function isResourceName(name)
    for i = 1, #resourceKeywords do
        if string.find(name, resourceKeywords[i], 1, true) then
            return true
        end
    end
    return false
end

local function scanResources(range)
    local list = {}
    if not root then return list end
    local charPos = root.Position
    local rangeSq = range * range

    local function tryAdd(model)
        if not model:IsA("Model") then return end
        local eid = model:GetAttribute("EntityID")
        if not eid then return end
        local pp = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
        if not pp then return end
        local delta = pp.Position - charPos
        local distSq = delta.X * delta.X + delta.Y * delta.Y + delta.Z * delta.Z
        if distSq <= rangeSq then
            table.insert(list, {eid = eid, distSq = distSq})
        end
    end

    local resources = workspace:FindFirstChild("Resources")
    if resources then
        for _, child in ipairs(resources:GetChildren()) do
            tryAdd(child)
            if child:IsA("Folder") and child.Name == "Breakaway" then
                for _, desc in ipairs(child:GetChildren()) do
                    tryAdd(desc)
                end
            end
        end
    end

    for _, child in ipairs(workspace:GetChildren()) do
        if child:IsA("Model") and isResourceName(child.Name) then
            tryAdd(child)
            local breakaway = child:FindFirstChild("Breakaway")
            if breakaway then
                for _, desc in ipairs(breakaway:GetChildren()) do
                    tryAdd(desc)
                end
            end
        elseif child:IsA("Folder") and child.Name == "Breakaway" then
            for _, desc in ipairs(child:GetChildren()) do
                tryAdd(desc)
            end
        end
    end

    table.sort(list, function(a, b) return a.distSq < b.distSq end)
    return list
end

local function startResource()
    if resourceActive then return end
    resourceActive = true

    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then
            if resourceConn then resourceConn:Disconnect() end
            resourceConn = nil
            resourceActive = false
            return
        end

        if not root or not GameUtil.Data.equipped then return end

        local ct = tick()
        local cooldown = tonumber(Options.resourcecooldownslider.Value) or 0.1
        if ct - lastResourceTime < cooldown then return end

        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1

        -- Rescan only every 0.45s instead of every swing
        if ct - lastResourceScan > 0.45 or #cachedTargets == 0 then
            cachedTargets = scanResources(range)
            lastResourceScan = ct
        end

        if #cachedTargets == 0 then return end

        local sel = {}
        local limit = math.min(tc, #cachedTargets)
        for i = 1, limit do
            sel[i] = cachedTargets[i].eid
        end

        swingtool(sel)
        lastResourceTime = ct
    end)
end

Options.resourceauratoggle:OnChanged(function(s)
    if s then
        startResource()
    else
        resourceActive = false
        if resourceConn then
            resourceConn:Disconnect()
            resourceConn = nil
        end
        table.clear(cachedTargets)
    end
end)

-- ========= [ DEPLOYABLE AURA FROM V9 ] =========
local RS = runs
local isTraderActive = false

local deployableAuraActive = false
local deployableAuraConnection = nil
local lastDeployableSwingTime = 0

function StartDeployableAura(range)
    if deployableAuraActive then return end
    deployableAuraActive = true
    range = range or 25

    deployableAuraConnection = RS.Heartbeat:Connect(function()
        if not deployableAuraActive or isTraderActive then return end

        local currentTime = tick()
        if currentTime - lastDeployableSwingTime < 0.1 then return end

        local char = plr.Character
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end

        local targetList = {}
        local folders = {"Deployables", "Rubble", "dropChests"}
      
        for _, folderName in ipairs(folders) do
            local folder = workspace:FindFirstChild(folderName)
            if folder then
                for _, child in pairs(folder:GetChildren()) do
                    local eid = child:GetAttribute("EntityID")
            
                    if eid then
                        local pp = child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart")
                        if pp and (pp.Position - rootPart.Position).Magnitude <= range then
                            table.insert(targetList, {entityID = eid, buffer = 0})
                        end
                    end
                end
            end
        end

        if #targetList > 0 then
            pcall(function()
                packets.SwingTool.send({
                    entityIDs = targetList,
                    cframe = rootPart.CFrame,
                    timestamp = clock.getServerTime(true)
                })
            end)
            lastDeployableSwingTime = currentTime
        end
    end)
end

function StopDeployableAura()

-- Critter Aura (optimized)
local SchuylerActive, critterConn = false, nil
local lastCritterTime, lastCritterScan = 0, 0
local cachedCritterTargets = {}

local function scanCritters(range)
    local list = {}
    if not root then return list end
    local charPos = root.Position
    local rangeSq = range * range
    local folder = workspace:FindFirstChild("Critters")
    if not folder then return list end

    for _, c in ipairs(folder:GetChildren()) do
        if c:IsA("Model") then
            local eid = c:GetAttribute("EntityID")
            if eid then
                local pp = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
                if pp then
                    local delta = pp.Position - charPos
                    local distSq = delta.X * delta.X + delta.Y * delta.Y + delta.Z * delta.Z
                    if distSq <= rangeSq then
                        table.insert(list, {eid = eid, distSq = distSq})
                    end
                end
            end
        end
    end

    table.sort(list, function(a, b) return a.distSq < b.distSq end)
    return list
end

    if SchuylerActive then return end

local function dropAllInventoryFast()

        and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")

    local groups = {}
    for _, child in ipairs(inv:GetChildren()) do
        if child:IsA("ImageLabel") and typeof(child.LayoutOrder) == "number" and child.Name ~= "" then
            local name = child.Name
            if not groups[name] then groups[name] = {} end
            table.insert(groups[name], {
                slot = child.LayoutOrder,
                qty = getSlotQuantity(child)
            })
        end
    end

    for _, entries in pairs(groups) do
        for _, entry in ipairs(entries) do
            local n = entry.qty
            if n < 1 then n = 1 end
            for i = 1, n do
                if i % 40 == 0 then
                    if not (Options.dropalltoggle and Options.dropalltoggle.Value) then return end
                    runs.Heartbeat:Wait()
                end
                packets.DropBagItem.send(entry.slot)
            end
            if not (Options.dropalltoggle and Options.dropalltoggle.Value) then return end
        end
    end
end

local lastPos = Vector3.new()
local stuckTimer = 0
local lastCheck = 0

local function walkToTarget(targetPosition)
    if hum and root and targetPosition then
        hum:MoveTo(targetPosition)
        local now = tick()
        local dt = now - lastCheck
        if lastCheck == 0 or dt > 0.5 then
            lastCheck = now
            lastPos = root.Position
            stuckTimer = 0
            return
        end
        lastCheck = now
        if (root.Position - targetPosition).Magnitude > 3 then
            if (root.Position - lastPos).Magnitude < 1 * dt then
                stuckTimer = stuckTimer + dt
                if stuckTimer >= 3 then
                    hum.Jump = true
                    stuckTimer = 0
                end
            else
                stuckTimer = 0
            end
        else
            stuckTimer = 0
        end
        lastPos = root.Position
    end
end

local function pathToPlantbox(range)
    local targetPos = getClosestPlantBoxPos(range)
    if targetPos then walkToTarget(targetPos) end
end

local function pathToBushesOrBoxes(range, fruitname)
    local targetPos = getClosestBushPos(range, fruitname)
    if targetPos then
        walkToTarget(targetPos)
    else
        local boxTarget = getClosestPlantBoxPos(range)
        if boxTarget then walkToTarget(boxTarget) end
    end
end

-- Table to remember what we've already picked up so we don't spam the server
local recentlyHarvested = {}

task.spawn(function()
    while true do
        if not Options.harvesttoggle.Value then
            task.wait(0.1)
            continue
        end

        local harvestrange = tonumber(Options.harvestrange.Value) or 30
        local selectedfruit = Options.fruitdropdown.Value
        local bushes = getbushes(harvestrange, selectedfruit)

        if bushes and #bushes > 0 then
            table.sort(bushes, function(a, b) return a.dist < b.dist end)
            
            local actionsThisFrame = 0
            
            for _, bush in ipairs(bushes) do
                if not Options.harvesttoggle.Value then break end
                
                -- Check if we already picked this bush up recently
                if not recentlyHarvested[bush.entityid] then
                    pickup(bush.entityid)
                    recentlyHarvested[bush.entityid] = os.clock() -- Mark it as harvested with the current time
                    
                    actionsThisFrame = actionsThisFrame + 1
                    
                    -- Every 15 bushes, wait a single frame (task.wait())
                    -- This prevents game freezing on massive clusters (50+ bushes) 
                    -- while still completing the entire batch in less than 0.1 seconds.
                    if actionsThisFrame % 15 == 0 then
                        task.wait() 
                    end
                end
            end
        end

        -- Clean up the memory table so it doesn't grow infinitely
        -- Removes bushes from the blacklist after 3 seconds (in case bushes respawn with the same ID)
        local currentTime = os.clock()
        for id, timeHarvested in pairs(recentlyHarvested) do
            if currentTime - timeHarvested > 0.1 then
                recentlyHarvested[id] = nil
            end
        end

        task.wait(0.1) -- Fast loop is back!
    end
end)

     local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y+100, sp.Z), Vector3.new(0,-200,0), rp)

local function findAllCauldrons()
    local dep = workspace:FindFirstChild("Deployables")
    if not dep or not (plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")) then return {} end
    local rp = plr.Character.HumanoidRootPart.Position
    local cauldrons = {}
    for _, obj in ipairs(dep:GetChildren()) do
        if obj.Name == "Cauldron" and obj:FindFirstChild("Hitbox") and (obj.Hitbox.Position - rp).Magnitude < 30 then
            table.insert(cauldrons, {cauldron = obj, hitbox = obj.Hitbox})
        end
    end
    return cauldrons
end

local function teleportToHitbox(hitbox)
    if not hitbox or not workspace:FindFirstChild("Items") then return end
    local tp = hitbox.Position + Vector3.new(math.random(-2,2)*0.3, 0.8, math.random(-2,2)*0.3)
    for _, item in ipairs(workspace.Items:GetChildren()) do
        local n = item.Name or ""
        if n:find("Ice Cube") or n:find("Bloodfruit") or n:find("Strawberry") or n:find("Prickly Pear") or n:find("Crystal") or n:find("Cloudberry") or n:find("Adurite") or n:find("Fire Hide") or n:find("Berry") or n:find("Iron") or n:find("Lemon") or n:find("Magnetite") then
            pcall(function()
                 if item:IsA("BasePart") then item.Position = tp
                elseif item:IsA("Model") and item.PrimaryPart then item:SetPrimaryPartCFrame(CFrame.new(tp)) end
            end)
        end
    end
end

local function brewPotions()
    if isBrewing then return end
    isBrewing = true
    local cauldrons = findAllCauldrons()
    if #cauldrons == 0 then isBrewing = false; return end
    local recipe = POTIONS[selectedPotion]
    if not recipe then isBrewing = false; return end
    for _, ing in ipairs(recipe) do
        local slot = getInventorySlot(ing.name)
        if slot then
            for _, c in ipairs(cauldrons) do dropFromSlot(slot, ing.amount); task.wait(0.5); teleportToHitbox(c.hitbox); task.wait(0.4) end
        end
        task.wait(0.8)
    end
    task.wait(2)
    for _, c in ipairs(cauldrons) do
        local eid = c.cauldron:GetAttribute("EntityID")
        if eid then pcall(function() packets.SweepPotion.send(eid) end) end
    end
    isBrewing = false
end

local function pressBedButton()
    local spawnGui = plr.PlayerGui:FindFirstChild("SpawnGui")
    if spawnGui and spawnGui:FindFirstChild("Customization") then
        local bedButton = spawnGui.Customization:FindFirstChild("BedButton")
        if bedButton then

local function getPlayerLevel()
    local success, GameUtilLocal = pcall(function() return require(rs.Modules.GameUtil) end)
    if success and GameUtilLocal.Data and GameUtilLocal.Data.level then return GameUtilLocal.Data.level end
    return plr:GetAttribute("Level") or 0
end

local function doRebirth()
    local success, packetsLocal = pcall(function() return require(rs.Modules.Packets) end)
    if success and packetsLocal.Rebirth then pcall(function() packetsLocal.Rebirth.send() end) end
    task.wait(2.5)
    pressBedButton()
end

local function AutoRebirthLoop()
    while autoRebirthRunning do
        task.wait(0.5)
        local spawnGui = plr.PlayerGui:FindFirstChild("SpawnGui")
        if spawnGui and spawnGui.Enabled then pressBedButton() end
        local currentLevel = getPlayerLevel()
        if currentLevel >= 100 then doRebirth(); task.wait(6) end
    end
end

local function StartAutoRebirth()
    if autoRebirthRunning then return end
    autoRebirthRunning = true
    task.spawn(AutoRebirthLoop)
end

local function StopAutoRebirth() autoRebirthRunning = false end

-- Tiered Target Verification Helper
local function getPrioritizedTarget()
    local critters = workspace:FindFirstChild("Critters")
    if not critters or not root then return nil end
    
    local traderInstance, queenInstance, minionInstance, antInstance = nil, nil, nil, nil
    local traderMinDist, queenMinDist, minionMinDist, antMinDist = math.huge, math.huge, math.huge, math.huge
    
    local children = critters:GetChildren()
    for i = 1, #children do
        local c = children[i]
        if c:IsA("Model") and c:GetAttribute("EntityID") then
            local pp = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
            if pp then
                local humCheck = c:FindFirstChildWhichIsA("Humanoid")
                if not humCheck or humCheck.Health > 0 then
                    local dist = (pp.Position - root.Position).Magnitude
                    
                    -- Priority 0: Wandering Trader Event Logic
                    if c.Name == "Wandering Trader" then
                        if dist < traderMinDist then
                            traderMinDist = dist
                            traderInstance = c
                        end
                    -- Priority 1: The Voodoo Spider Queen
                    elseif c.Name == "The Voodoo Spider Queen" and not traderInstance then
                        if dist < queenMinDist then
                            queenMinDist = dist
                            queenInstance = c
                        end
                    -- Priority 2: Soul Minion
                    elseif c.Name == "Soul Minion" and not traderInstance and not queenInstance then
                        if dist < minionMinDist then
                            minionMinDist = dist
                            minionInstance = c
                        end
                    -- Priority 3: Void Ant
                     elseif c.Name == "Void Ant" and not traderInstance and not queenInstance and not minionInstance then
                        if dist < antMinDist then
                            antMinDist = dist
                            antInstance = c
                        end
                    end
                end
            end
        end
     end
    
    if traderInstance then return traderInstance end
    if queenInstance then return queenInstance end
    if minionInstance then return minionInstance end
    if antInstance then return antInstance end
    return nil
end

-- Completely Rewritten Lockout Execution Loop
task.spawn(function()
    local currentlyLockedID = nil
    local targetExpiryDelay = 0
    
    while true do
        if Options.tpsoulminionstoggle and Options.tpsoulminionstoggle.Value and root then
            local rightNow = tick()
             
            -- Firmly enforce the 3-second delay period after any target vanishes
            if rightNow < targetExpiryDelay then
                task.wait(0.05)
                continue
            end
             
            -- If we are currently tracking an ID, make sure it still exists in workspace
            local activeTargetStillAlive = false
            if currentlyLockedID then
                local critters = workspace:FindFirstChild("Critters")
                if critters then
                    for _, c in ipairs(critters:GetChildren()) do
                        if c:GetAttribute("EntityID") == currentlyLockedID then
                            local humCheck = c:FindFirstChildWhichIsA("Humanoid")
                            if not humCheck or humCheck.Health > 0 then
                                activeTargetStillAlive = true
                                local targetPart = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
                                if targetPart then
                                    -- Continuous strict position lock teleportation
                                    root.CFrame = targetPart.CFrame + Vector3.new(0, 3, 0)
                                end
                                break
                            end
                        end
                    end
                end
                
                -- TARGET ELIMINATED: If it was alive last frame but is now gone/dead, initiate the 3-second lockout delay
                 if not activeTargetStillAlive then
                    print("[Exploit Engine] Target eliminated!\nTriggering strict 3-second swap delay.")
                    targetExpiryDelay = rightNow + 1.5
                    currentlyLockedID = nil
                end
            end
            
             -- If we aren't tracking anything, look for the next prioritized target
            if not currentlyLockedID and rightNow >= targetExpiryDelay then
                local nextTarget = getPrioritizedTarget()
                if nextTarget then
                    currentlyLockedID = nextTarget:GetAttribute("EntityID")
                 end
            end
        else
            currentlyLockedID = nil
         end
        task.wait(0.02)
    end
end)
