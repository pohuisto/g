local function fetchWithRetry(url)
    while true do
        local ok, res = pcall(function()
            return game:HttpGetAsync(url)
        end)
        if ok and type(res) == "string" and #res > 50 then
            return res
        end
        warn("[Loader] HttpGetAsync failed/timed out, retrying in 2s...", tostring(res))
        task.wait(2)
    end
end

-- Helper function to format seconds into HH:MM:SS
local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- Robust inventory scanner that checks background data modules and UI layouts
local function getInventoryCount(itemName)
    if GameUtil and GameUtil.Data then
        local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
        if type(inv) == "table" then
            if inv[itemName] then return tonumber(inv[itemName]) or 0 end
            for _, v in pairs(inv) do
                if type(v) == "table" and (v.name == itemName or v.Name == itemName) then
                    return tonumber(v.quantity or v.amount or v.Amount or v.count or v.Count or 1) or 0
                end
            end
        end
    end
    
    local list = plr.PlayerGui:FindFirstChild("MainGui") 
        and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") 
        and plr.PlayerGui:FindFirstChild("MainGui").RightPanel:FindFirstChild("Inventory") 
        and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if list then
        local total = 0
        for _, slot in ipairs(list:GetChildren()) do
            if slot:IsA("GuiObject") and (slot.Name == itemName or slot.Name:find(itemName)) then
                local quantityImg = slot:FindFirstChild("QuantityImage")
                local textObj = (quantityImg and quantityImg:FindFirstChild("QuantityText")) or slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChildWhichIsA("TextLabel")
                if textObj then
                    total = total + (tonumber(textObj.Text:match("%d+")) or 1)
                else
                    total = total + 1
                end
            end
        end
        if total > 0 then return total end
    end
    return 0
end

local killAuraActive, killAuraConn, lastKillTime = false, nil, 0

-- Fixed Kill Aura (matches real packet format)
local function startKillAura()
    if killAuraActive then return end
    killAuraActive = true

    killAuraConn = runs.Heartbeat:Connect(function()
        if not Options.killauratoggle.Value then
            if killAuraConn then killAuraConn:Disconnect() end
            killAuraConn = nil
            killAuraActive = false
            return
        end

        if not root or not GameUtil.Data.equipped then return end

        local ct = tick()
        local cooldown = tonumber(Options.killauracooldown.Value) or 0.12
        if ct - lastKillTime < cooldown then return end

        local range = tonumber(Options.killaurarange.Value) or 20
        local targets = {}

        for _, player in ipairs(game.Players:GetPlayers()) do
            if player ~= plr and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local dist = (hrp.Position - root.Position).Magnitude
                    if dist <= range then
                        local eid = player.Character:GetAttribute("EntityID")
                        if eid then
table.insert(targets, eid)
                        end
                    end
                end
            end
        end

        if #targets > 0 then
            pcall(function()
                packets.SwingTool.send({
                    entityIDs = targets,
                    cframe = root.CFrame,
                    timestamp = clock.getServerTime(true)
                })
            end)
            lastKillTime = ct
        end
    end)
end

Options.killauratoggle:OnChanged(function(state)
    if state then
        startKillAura()
    else
        killAuraActive = false
        if killAuraConn then
            killAuraConn:Disconnect()
            killAuraConn = nil
        end
    end
end)

local function getFishTrapsWithFish()
    local traps = {}
    local dep = workspace:FindFirstChild("Deployables")
    if not (dep and root) then return traps end

    local range = tonumber(Options.fishtraprangeslider.Value) or 300
    local rangeSq = range * range
    local charPos = root.Position

    for _, trap in ipairs(dep:GetChildren()) do
        if trap.Name == "Fish Trap" then
            local contents = trap:FindFirstChild("Contents")
            if contents then
                for _, item in ipairs(contents:GetChildren()) do
                    if item.Name == "Raw Fish" then
                        local eid = item:GetAttribute("EntityID")
                        if eid then
                            local pp = trap.PrimaryPart or trap:FindFirstChildWhichIsA("BasePart")
                            if pp then
                                local delta = pp.Position - charPos
                                local distSq = delta.X*delta.X + delta.Y*delta.Y + delta.Z*delta.Z
                                if distSq <= rangeSq then
                                    traps[#traps+1] = {
                                        trap = trap,
                                        part = pp,
                                        entityID = eid,
                                        distSq = distSq
                                    }
                                end
                            end
                        end
                        break
                    end
                end
            end
        end
    end

    table.sort(traps, function(a, b) return a.distSq < b.distSq end)
    return traps
end

task.spawn(function()
    local lastTeleport = 0
    local TELEPORT_COOLDOWN = 0.01

    while true do
        if Options.autofishtraptoggle and Options.autofishtraptoggle.Value and root then
            local traps = getFishTrapsWithFish()

            if #traps > 0 then
                local target = traps[1]
                local now = tick()

                if now - lastTeleport >= TELEPORT_COOLDOWN then
                    local offset = Vector3.new(
                        math.random(-8, 8) * 0.1,
                        4 + math.random() * 0.6,
                        math.random(-8, 8) * 0.1
                    )
                    root.CFrame = CFrame.new(target.part.Position + offset)
                    lastTeleport = now
                    task.wait(0.15)
                end
            else
                task.wait(0.45)
            end
        else
            task.wait(0.3)
        end

        task.wait(0.04)
    end
end)

local noclipPartConn
local function updPartNoclip()
    local anyOn = false
    for _, g in ipairs(noclipPartGroups) do
        if Options[g.opt] and Options[g.opt].Value then anyOn = true break end
    end
    if Options.noclipaccessoriestoggle and Options.noclipaccessoriestoggle.Value then anyOn = true end

    if not anyOn then
        if noclipPartConn then noclipPartConn:Disconnect(); noclipPartConn = nil end
        return
    end
    if noclipPartConn then return end

    noclipPartConn = runs.Stepped:Connect(function()
        local c = plr.Character
        if not c then return end
        local accOn = Options.noclipaccessoriestoggle and Options.noclipaccessoriestoggle.Value
        for _, part in ipairs(c:GetDescendants()) do
            if part:IsA("BasePart") then
                local opt = noclipPartLookup[part.Name]
                if opt and Options[opt] and Options[opt].Value then
                    part.CanCollide = false
                elseif accOn and part.Parent and part.Parent:IsA("Accessory") then
                    part.CanCollide = false
                end
            end
        end
    end)
end
for _, g in ipairs(noclipPartGroups) do
    Options[g.opt]:OnChanged(updPartNoclip)
end
Options.noclipaccessoriestoggle:OnChanged(updPartNoclip)

-- Free Cam (client-only camera override, zero character movement / zero flagging)
do
    local uis = game:GetService("UserInputService")
    local cas = game:GetService("ContextActionService")
    local fcActive = false
    local fcStepName = "AntigravityFreeCamStep"
    local fcSinkName = "AntigravityFreeCamSink"
    local fcPos = Vector3.zero
    local fcYaw, fcPitch = 0, 0
    local fcInputConn = nil
    local fcDragging = false

    local function stopFreeCam()
        if not fcActive then return end
        fcActive = false
        fcDragging = false
        pcall(function() runs:UnbindFromRenderStep(fcStepName) end)
        pcall(function() cas:UnbindAction(fcSinkName) end)
        if fcInputConn then fcInputConn:Disconnect(); fcInputConn = nil end
        pcall(function() uis.MouseBehavior = Enum.MouseBehavior.Default end)
        local cam = workspace.CurrentCamera
        if cam then
            cam.CameraType = Enum.CameraType.Custom
            local h = plr.Character and plr.Character:FindFirstChildOfClass("Humanoid")
            if h then cam.CameraSubject = h end
        end
    end

    local function startFreeCam()
        if fcActive then return end
        local cam = workspace.CurrentCamera
        if not cam then return end
        fcActive = true
        fcPos = cam.CFrame.Position
        local rx, ry, _ = cam.CFrame:ToEulerAnglesYXZ()
        fcPitch = rx
        fcYaw = ry

        cam.CameraType = Enum.CameraType.Scriptable

        pcall(function()
            cas:BindActionAtPriority(
                fcSinkName,
                function() return Enum.ContextActionResult.Sink end,
                false,
                Enum.ContextActionPriority.High.Value + 500,
                Enum.KeyCode.W, Enum.KeyCode.A, Enum.KeyCode.S, Enum.KeyCode.D, Enum.KeyCode.Space
            )
        end)

        fcInputConn = uis.InputChanged:Connect(function(input)
            if not fcActive then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement and uis:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                fcYaw = fcYaw - input.Delta.X * 0.004
                fcPitch = math.clamp(fcPitch - input.Delta.Y * 0.004, -1.52, 1.52)
            end
        end)

        runs:BindToRenderStep(fcStepName, Enum.RenderPriority.Camera.Value + 1, function(dt)
            if not fcActive then return end
            local c = workspace.CurrentCamera
            if not c then return end
            if c.CameraType ~= Enum.CameraType.Scriptable then
                c.CameraType = Enum.CameraType.Scriptable
            end

            if uis:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                if not fcDragging then
                    fcDragging = true
                    pcall(function() uis.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition end)
                end
            elseif fcDragging then
                fcDragging = false
                pcall(function() uis.MouseBehavior = Enum.MouseBehavior.Default end)
            end

            local rot = CFrame.fromEulerAnglesYXZ(fcPitch, fcYaw, 0)
            local move = Vector3.zero
            if not uis:GetFocusedTextBox() then
                if uis:IsKeyDown(Enum.KeyCode.W) then move = move + rot.LookVector end
                if uis:IsKeyDown(Enum.KeyCode.S) then move = move - rot.LookVector end
                if uis:IsKeyDown(Enum.KeyCode.D) then move = move + rot.RightVector end
                if uis:IsKeyDown(Enum.KeyCode.A) then move = move - rot.RightVector end
                if uis:IsKeyDown(Enum.KeyCode.E) or uis:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0, 1, 0) end
                if uis:IsKeyDown(Enum.KeyCode.Q) or uis:IsKeyDown(Enum.KeyCode.LeftControl) then move = move - Vector3.new(0, 1, 0) end
            end

            local speed = (Options.freecamspeed and tonumber(Options.freecamspeed.Value)) or 50
            if uis:IsKeyDown(Enum.KeyCode.LeftShift) then speed = speed * 2.5 end
            if move.Magnitude > 0 then
                fcPos = fcPos + move.Unit * speed * dt
            end
            c.CFrame = CFrame.new(fcPos) * rot
        end)
    end

    Options.freecamtoggle:OnChanged(function(state)
        if state then startFreeCam() else stopFreeCam() end
    end)
end

-- FPS Boost
do
    local lighting = game:GetService("Lighting")
    local savedLighting = nil
    local savedEffects = {}

    local function applyFpsBoost(on)
        if on then

                lighting.FogEnd = 9e9
                lighting.Brightness = 1
            end)
            pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)

            savedEffects = {}
            for _, v in ipairs(lighting:GetChildren()) do
                if v:IsA("PostEffect") then
                    savedEffects[v] = v.Enabled
                    v.Enabled = false
                end
            end

            if ter then
                savedTerrain = {
                    WaterWaveSize = ter.WaterWaveSize,
                    WaterWaveSpeed = ter.WaterWaveSpeed,
                    WaterReflectance = ter.WaterReflectance
                }
                pcall(function()
                    ter.WaterWaveSize = 0
                    ter.WaterWaveSpeed = 0
                    ter.WaterReflectance = 0
                end)
                pcall(function()
                    if sethiddenproperty then sethiddenproperty(ter, "Decoration", false) end
                end)
            end

            savedParticles = {}
            for _, obj in ipairs(workspace:GetDescendants()) do
                if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
                    savedParticles[obj] = obj.Enabled
                    obj.Enabled = false
                end
            end
        else
            if savedLighting then
                pcall(function()
                    lighting.GlobalShadows = savedLighting.GlobalShadows
                    lighting.FogEnd = savedLighting.FogEnd
                    lighting.Brightness = savedLighting.Brightness
                end)
                savedLighting = nil
            end
            for v, was in pairs(savedEffects) do
                if v and v.Parent then pcall(function() v.Enabled = was end) end
            end
            savedEffects = {}
            local ter = workspace:FindFirstChildOfClass("Terrain")
            if ter and savedTerrain then
                pcall(function()
                    ter.WaterWaveSize = savedTerrain.WaterWaveSize
                    ter.WaterWaveSpeed = savedTerrain.WaterWaveSpeed
                    ter.WaterReflectance = savedTerrain.WaterReflectance
                end)
                pcall(function()
                    if sethiddenproperty then sethiddenproperty(ter, "Decoration", true) end
                end)
                savedTerrain = nil
            end
            for obj, was in pairs(savedParticles) do
                if obj and obj.Parent then pcall(function() obj.Enabled = was end) end
            end
            savedParticles = {}
        end
    end

    Options.fpsboosttoggle:OnChanged(function(state)
        applyFpsBoost(state)
    end)
end

local resourceActive, resourceConn = false, nil
local lastResourceTime, lastResourceScan = 0, 0
local cachedTargets = {}

local resourceKeywords = {"Node", "Stone", "Lode", "Rock", "Coal", "Guardian", "Chunk", "Ore"}

local function isResourceName(name)
    for i = 1, #resourceKeywords do
        if string.find(name, resourceKeywords[i], 1, true) then
            return true
        end
    end
    return false
end

local function scanResources(range)
    local list = {}
    if not root then return list end
    local charPos = root.Position
    local rangeSq = range * range

    local function tryAdd(model)
        if not model:IsA("Model") then return end
        local eid = model:GetAttribute("EntityID")
        if not eid then return end
        local pp = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
        if not pp then return end
        local delta = pp.Position - charPos
        local distSq = delta.X * delta.X + delta.Y * delta.Y + delta.Z * delta.Z
        if distSq <= rangeSq then
            table.insert(list, {eid = eid, distSq = distSq})
        end
    end

    local resources = workspace:FindFirstChild("Resources")
    if resources then
        for _, child in ipairs(resources:GetChildren()) do
            tryAdd(child)
            if child:IsA("Folder") and child.Name == "Breakaway" then
                for _, desc in ipairs(child:GetChildren()) do
                    tryAdd(desc)
                end
            end
        end
    end

    for _, child in ipairs(workspace:GetChildren()) do
        if child:IsA("Model") and isResourceName(child.Name) then
            tryAdd(child)
            local breakaway = child:FindFirstChild("Breakaway")
            if breakaway then
                for _, desc in ipairs(breakaway:GetChildren()) do
                    tryAdd(desc)
                end
            end
        elseif child:IsA("Folder") and child.Name == "Breakaway" then
            for _, desc in ipairs(child:GetChildren()) do
                tryAdd(desc)
            end
        end
    end

    table.sort(list, function(a, b) return a.distSq < b.distSq end)
    return list
end

local function startResource()
    if resourceActive then return end
    resourceActive = true

    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then
            if resourceConn then resourceConn:Disconnect() end
            resourceConn = nil
            resourceActive = false
            return
        end

if not root then return end

        local ct = tick()
        local cooldown = tonumber(Options.resourcecooldownslider.Value) or 0.1
        if ct - lastResourceTime < cooldown then return end

        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1

        -- Rescan only every 0.45s instead of every swing
        if ct - lastResourceScan > 0.45 or #cachedTargets == 0 then
            cachedTargets = scanResources(range)
            lastResourceScan = ct
        end

        if #cachedTargets == 0 then return end

        local sel = {}
        local limit = math.min(tc, #cachedTargets)
        for i = 1, limit do
            sel[i] = cachedTargets[i].eid
        end

        swingtool(sel)
        lastResourceTime = ct
    end)
end

Options.resourceauratoggle:OnChanged(function(s)
    if s then
        startResource()
    else
        resourceActive = false
        if resourceConn then
            resourceConn:Disconnect()
            resourceConn = nil
        end
        table.clear(cachedTargets)
    end
end)

-- PICKUP — blast all, after 5s stuck spam max speed
local STUCK_TIME = 5
local RETRY = 0.15

local tracked = {}
local tried = {}
local stuckSince = 0
local itemsFolder = nil
local hookConnAdd, hookConnRem
local building = false

local function shouldPickup(name)
    if #selecteditems == 0 then return true end
    return table.find(selecteditems, name) ~= nil
end

local function track(item)
    if not (item:IsA("BasePart") or item:IsA("MeshPart")) then return end
    if not shouldPickup(item.Name) then return end
    local eid = item:GetAttribute("EntityID")
    if eid then tracked[item] = eid end
end

local function untrack(item)
    tracked[item] = nil
end

local function hookFolder(folder)
    if hookConnAdd then hookConnAdd:Disconnect() end
    if hookConnRem then hookConnRem:Disconnect() end
    itemsFolder = folder
    if not folder then return end

    hookConnAdd = folder.ChildAdded:Connect(track)
    hookConnRem = folder.ChildRemoved:Connect(untrack)

    if building then return end
    building = true
    task.spawn(function()
        table.clear(tracked)
        local children = folder:GetChildren()
        for i = 1, #children do
            track(children[i])
            if i % 200 == 0 then task.wait() end
        end
        building = false
    end)
end

task.spawn(function()
    while true do
        local f = workspace:FindFirstChild("Items")
        if f ~= itemsFolder then
            hookFolder(f)
        end
        task.wait(1)
    end
end)

Options.itemdropdown:OnChanged(function(v)
    selecteditems = {}
    for item, state in pairs(v) do
        if state then table.insert(selecteditems, item) end
    end
    if itemsFolder then hookFolder(itemsFolder) end
end)

task.spawn(function()
    while true do
        if not (Options.autopickuptoggle and Options.autopickuptoggle.Value and root) then
            stuckSince = 0
            task.wait(0.12)
            continue
        end

        local now = tick()
        local range = tonumber(Options.pickuprange.Value) or 35
        local rangeSq = range * range
        local pos = root.Position

        local inRange = {}
        for part, eid in pairs(tracked) do
            if not part.Parent then
                tracked[part] = nil
            else
                local p = part.Position
                local dx, dy, dz = p.X - pos.X, p.Y - pos.Y, p.Z - pos.Z
                if dx*dx + dy*dy + dz*dz <= rangeSq then
                    inRange[#inRange + 1] = eid
                end
            end
        end

        if #inRange == 0 then
            stuckSince = 0
            task.wait(0.05)
            continue
        end

        if stuckSince == 0 then
            stuckSince = now
        end

        local spam = (now - stuckSince) >= STUCK_TIME

        if spam then
            -- MAX SPEED: every item every frame, no retry lock
            for i = 1, #inRange do
                pickup(inRange[i])
            end
            runs.Heartbeat:Wait()
        else
            -- normal blast pulse
            for i = 1, #inRange do
                local eid = inRange[i]
                local prev = tried[eid]
                if not prev or now - prev >= RETRY then
                    pickup(eid)
                    tried[eid] = now
                end
            end
            task.wait(0.03)
        end

        if now % 3 < 0.05 then
            for eid, t in pairs(tried) do
                if now - t > 1.2 then tried[eid] = nil end
            end
        end
    end
end)

local function getSlotQuantity(slot)
    local quantityImg = slot:FindFirstChild("QuantityImage")
    local textObj = (quantityImg and quantityImg:FindFirstChild("QuantityText"))
        or slot:FindFirstChild("Amount")
        or slot:FindFirstChild("Count")
        or slot:FindFirstChildWhichIsA("TextLabel")
    if textObj and textObj.Text then
        local txt = string.lower(textObj.Text)
        local num = tonumber(txt:match("[%d%.]+"))
        if not num then return 1 end
        if txt:find("k", 1, true) then num = num * 1000
        elseif txt:find("m", 1, true) then num = num * 1000000 end
        return math.max(1, math.floor(num))
    end
    return 1
end

-- AUTO DROP ALL (fast + low lag)
-- Every frame: look at all occupied slots once, then send a fixed budget of drop packets
-- spread round-robin over ALL slots (so every item type drains at the same time),
-- then wait exactly one Heartbeat. No giant per-slot loops, no long yields.
local DROPALL_BUDGET = 60 -- max drop packets per frame

local dropAllList = nil
local function getDropAllList()
    if dropAllList and dropAllList.Parent then return dropAllList end
    local pg = plr:FindFirstChild("PlayerGui")
    local mg = pg and pg:FindFirstChild("MainGui")
    local rp = mg and mg:FindFirstChild("RightPanel")
    local invf = rp and rp:FindFirstChild("Inventory")
    dropAllList = invf and invf:FindFirstChild("List") or nil
    return dropAllList
end

local function dropAllInventoryFast()
    local inv = getDropAllList()
    if not inv then return false end
    local sendDrop = packets.DropBagItem and packets.DropBagItem.send
    if not sendDrop then return false end

    local slots, remaining = {}, {}
    for _, child in ipairs(inv:GetChildren()) do
        if child:IsA("ImageLabel") and typeof(child.LayoutOrder) == "number" and child.Name ~= "" then
            slots[#slots + 1] = child.LayoutOrder
            remaining[#slots] = getSlotQuantity(child)
        end
    end
    if #slots == 0 then return false end

    local budget = DROPALL_BUDGET
    local progressed = true
    while budget > 0 and progressed do
        progressed = false
        for i = 1, #slots do
            if remaining[i] > 0 and budget > 0 then
                sendDrop(slots[i])
                remaining[i] = remaining[i] - 1
                budget = budget - 1
                progressed = true
            end
        end
    end
    return true
end

task.spawn(function()
    while true do
        if Options.dropalltoggle and Options.dropalltoggle.Value then
            pcall(dropAllInventoryFast)
            runs.Heartbeat:Wait()
        else
            task.wait(0.15)
        end
    end
end)

    local children = cropsFolder:GetChildren()

        end

local lastPos = Vector3.new()
local stuckTimer = 0
local lastCheck = 0

local function walkToTarget(targetPosition)
    if hum and root and targetPosition then
        hum:MoveTo(targetPosition)
        local now = tick()
        local dt = now - lastCheck
        if lastCheck == 0 or dt > 0.5 then
            lastCheck = now
            lastPos = root.Position
            stuckTimer = 0
            return
        end
        lastCheck = now
        if (root.Position - targetPosition).Magnitude > 3 then
            if (root.Position - lastPos).Magnitude < 1 * dt then
                stuckTimer = stuckTimer + dt
                if stuckTimer >= 3 then
                    hum.Jump = true
                    stuckTimer = 0
                end
            else
                stuckTimer = 0
            end
        else
            stuckTimer = 0
        end
        lastPos = root.Position
    end
end

local function pathToPlantbox(range)
    local targetPos = getClosestPlantBoxPos(range)
    if targetPos then walkToTarget(targetPos) end
end

local function pathToBushesOrBoxes(range, fruitname)
    local targetPos = getClosestBushPos(range, fruitname)
    if targetPos then
        walkToTarget(targetPos)
    else
        local boxTarget = getClosestPlantBoxPos(range)
        if boxTarget then walkToTarget(boxTarget) end
    end
end

                    -- If server confirms it has a seed, clear local debounce
                    if hasSeed and entityid then

                    elseif not hasSeed and entityid and not localPlantDebounce[entityid] then
                        local ppart = box.PrimaryPart or box:FindFirstChildWhichIsA("BasePart")
                        if ppart then
                            local dist = (ppart.Position - charPos).Magnitude

                                end
                            end
                        end
                    end
                end
            end
            
            -- Cleanup stale debounces (if planted but server packet dropped)
            local now = tick()
            for id, timePlanted in pairs(localPlantDebounce) do
                if now - timePlanted > 0.01 then
                    localPlantDebounce[id] = nil
                end
            end
        end

        runs.Heartbeat:Wait()

-- Table to remember what we've already picked up so we don't spam the server
local recentlyHarvested = {}

task.spawn(function()
    while true do
        if not Options.harvesttoggle.Value then
            task.wait(0.1)
            continue
        end

        local harvestrange = tonumber(Options.harvestrange.Value) or 30
        local selectedfruit = Options.fruitdropdown.Value
        local bushes = getbushes(harvestrange, selectedfruit)

        if bushes and #bushes > 0 then
            table.sort(bushes, function(a, b) return a.dist < b.dist end)
            
            local actionsThisFrame = 0
            
            for _, bush in ipairs(bushes) do
                if not Options.harvesttoggle.Value then break end
                
                -- Check if we already picked this bush up recently
                if not recentlyHarvested[bush.entityid] then
                    pickup(bush.entityid)
                    recentlyHarvested[bush.entityid] = os.clock() -- Mark it as harvested with the current time
                    
                    actionsThisFrame = actionsThisFrame + 1
                    
                    -- Every 15 bushes, wait a single frame (task.wait())
                    -- This prevents game freezing on massive clusters (50+ bushes) 
                    -- while still completing the entire batch in less than 0.1 seconds.
                    if actionsThisFrame % 15 == 0 then
                        task.wait() 
                    end
                end
            end
        end

        -- Clean up the memory table so it doesn't grow infinitely
        -- Removes bushes from the blacklist after 3 seconds (in case bushes respawn with the same ID)
        local currentTime = os.clock()
        for id, timeHarvested in pairs(recentlyHarvested) do
            if currentTime - timeHarvested > 0.1 then
                recentlyHarvested[id] = nil
            end
        end

        task.wait(0.1) -- Fast loop is back!
    end
end)

-- REAL FARMING STATISTICS (event-driven via Game.inventory + per-fruit fallback)
do
    local function recordHarvest(fruitName, amount)
        if not fruitName or not fruittoitemid[fruitName] then return end
        if not amount or amount <= 0 then return end
        -- When Auto Harvest is off, still count manual bush harvests (<= 25 per packet) without counting chest transfers
        local harvestingOn = Options.harvesttoggle and Options.harvesttoggle.Value
        if not harvestingOn and amount > 25 then return end

        if not farmStats.timerStarted then
            farmStats.timerStarted = true
            farmStats.startTime = tick()
            farmStats.sessionStartTime = tick()
        end
        farmStats.harvested[fruitName] = (farmStats.harvested[fruitName] or 0) + amount
        farmStats.totalAllTime = farmStats.totalAllTime + amount
    end

    local hookedInv = false
    pcall(function()
        local gameFolder = rs:FindFirstChild("Game")
        local invModScript = gameFolder and gameFolder:FindFirstChild("inventory")
        local invMod = invModScript and require(invModScript)
        if invMod and type(invMod.itemAdded) == "function" and type(invMod.itemChanged) == "function" then
            local origAdded = invMod.itemAdded
            invMod.itemAdded = function(name, qty, idx, ...)
                pcall(recordHarvest, name, tonumber(qty) or 0)
                return origAdded(name, qty, idx, ...)
            end

            local origChanged = invMod.itemChanged
            invMod.itemChanged = function(layoutOrder, newQty, ...)
                local inv = GameUtil and GameUtil.Data and GameUtil.Data.inventory
                local entry = type(inv) == "table" and inv[layoutOrder]
                local oldQty = (type(entry) == "table" and tonumber(entry.quantity)) or 0
                local delta = (tonumber(newQty) or 0) - oldQty
                if type(entry) == "table" and delta > 0 then
                    pcall(recordHarvest, entry.name, delta)
                end
                return origChanged(layoutOrder, newQty, ...)
            end
            hookedInv = true
        end
    end)

    -- Fallback polling per fruit only if inventory module hook wasn't available
    local lastCounts = {}
    task.spawn(function()
        while true do
            local targetFruit = (Options.fruitdropdown and Options.fruitdropdown.Value) or "Bloodfruit"
            if not hookedInv then
                local cur = getInventoryCount(targetFruit) or 0
                local prev = lastCounts[targetFruit]
                if prev == nil then
                    lastCounts[targetFruit] = cur
                else
                    if cur > prev then
                        recordHarvest(targetFruit, cur - prev)
                    end
                    lastCounts[targetFruit] = cur
                end
            end

            local count = farmStats.harvested[targetFruit] or 0
            local elapsed = farmStats.timerStarted and (tick() - farmStats.startTime) or 0
            local totalSessionElapsed = farmStats.timerStarted and (tick() - farmStats.sessionStartTime) or 0
            if elapsed < 1 then elapsed = 1 end

            local perMin = farmStats.timerStarted and math.round((count / elapsed) * 60) or 0
            local perHour = farmStats.timerStarted and math.round((count / elapsed) * 3600) or 0

            pcall(function()
                BloodfruitParagraph:SetContent(string.format("%s: %d (Per Min: %d |\nPer Hour: %d)", targetFruit, count, perMin, perHour))
                if farmStats.timerStarted then
                    TotalAllTimeParagraph:SetContent(string.format("%d (Time: %s)", farmStats.totalAllTime, formatTime(totalSessionElapsed)))
                else
                    TotalAllTimeParagraph:SetContent(string.format("%d (Time: 00:00:00 (Waiting for harvest...))", farmStats.totalAllTime))
                end
            end)

            task.wait(0.3)
        end
    end)
end

-- Tiered Target Verification Helper
local function getPrioritizedTarget()
    local critters = workspace:FindFirstChild("Critters")
    if not critters or not root then return nil end
    
    local traderInstance, queenInstance, minionInstance, antInstance = nil, nil, nil, nil
    local traderMinDist, queenMinDist, minionMinDist, antMinDist = math.huge, math.huge, math.huge, math.huge
    
    local children = critters:GetChildren()
    for i = 1, #children do
        local c = children[i]
        if c:IsA("Model") and c:GetAttribute("EntityID") then
            local pp = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
            if pp then
                local humCheck = c:FindFirstChildWhichIsA("Humanoid")
                if not humCheck or humCheck.Health > 0 then
                    local dist = (pp.Position - root.Position).Magnitude
                    
                    -- Priority 0: Wandering Trader Event Logic
                    if c.Name == "Wandering Trader" then
                        if dist < traderMinDist then
                            traderMinDist = dist
                            traderInstance = c
                        end
                    -- Priority 1: The Voodoo Spider Queen
                    elseif c.Name == "The Voodoo Spider Queen" and not traderInstance then
                        if dist < queenMinDist then
                            queenMinDist = dist
                            queenInstance = c
                        end
                    -- Priority 2: Soul Minion
                    elseif c.Name == "Soul Minion" and not traderInstance and not queenInstance then
                        if dist < minionMinDist then
                            minionMinDist = dist
                            minionInstance = c
                        end
                    -- Priority 3: Void Ant
                     elseif c.Name == "Void Ant" and not traderInstance and not queenInstance and not minionInstance then
                        if dist < antMinDist then
                            antMinDist = dist
                            antInstance = c
                        end
                    end
                end
            end
        end
     end
    
    if traderInstance then return traderInstance end
    if queenInstance then return queenInstance end
    if minionInstance then return minionInstance end
    if antInstance then return antInstance end
    return nil
end
